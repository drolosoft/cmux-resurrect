package setup

import (
	"fmt"
	"os"
	"path/filepath"
	"strings"
)

const hookMarker = "# crex-pop-hook"

// DetectShell returns the shell name from $SHELL (e.g. "zsh", "bash", "fish").
func DetectShell() string {
	shell := os.Getenv("SHELL")
	return filepath.Base(shell)
}

// HookLine returns the shell-specific keybinding line for crex pop.
// The key parameter is shell-specific: "^G" for zsh, "\\C-g" for bash, "\\cg" for fish.
func HookLine(shell, key string) string {
	switch shell {
	case "zsh":
		return fmt.Sprintf(`bindkey -s '%s' 'crex pop\n' %s`, key, hookMarker)
	case "bash":
		return fmt.Sprintf(`bind '"%s": "crex pop\n"' %s`, key, hookMarker)
	case "fish":
		return fmt.Sprintf(`bind %s 'crex pop; commandline -f repaint' %s`, key, hookMarker)
	default:
		return ""
	}
}

// DefaultKey returns the default key binding string for the given shell.
func DefaultKey(shell string) string {
	switch shell {
	case "zsh":
		return "^G"
	case "bash":
		return "\\C-g"
	case "fish":
		return "\\cg"
	default:
		return ""
	}
}

// RCFilePath returns the default rc file path for the given shell.
func RCFilePath(shell string) string {
	home, _ := os.UserHomeDir()
	switch shell {
	case "zsh":
		return filepath.Join(home, ".zshrc")
	case "bash":
		return filepath.Join(home, ".bashrc")
	case "fish":
		return filepath.Join(home, ".config", "fish", "config.fish")
	default:
		return ""
	}
}

// InstallHookToFile appends the hook line to an rc file if not already present.
func InstallHookToFile(path, shell, key string) error {
	content, err := os.ReadFile(path)
	if err != nil && !os.IsNotExist(err) {
		return fmt.Errorf("read %s: %w", path, err)
	}

	if strings.Contains(string(content), hookMarker) {
		return nil // already installed
	}

	line := HookLine(shell, key)
	if line == "" {
		return fmt.Errorf("unsupported shell: %s", shell)
	}

	f, err := os.OpenFile(path, os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0644)
	if err != nil {
		return fmt.Errorf("open %s: %w", path, err)
	}
	defer f.Close()

	_, err = fmt.Fprintf(f, "\n%s\n", line)
	return err
}

// UninstallHookFromFile removes the hook line from an rc file.
func UninstallHookFromFile(path string) error {
	content, err := os.ReadFile(path)
	if err != nil {
		return err
	}

	var lines []string
	for _, line := range strings.Split(string(content), "\n") {
		if !strings.Contains(line, hookMarker) {
			lines = append(lines, line)
		}
	}

	return os.WriteFile(path, []byte(strings.Join(lines, "\n")), 0644)
}

package tui

import (
	"fmt"
	"strings"

	tea "github.com/charmbracelet/bubbletea"
	"github.com/charmbracelet/lipgloss"
)

// PopItem represents a single entry in the pop picker.
type PopItem struct {
	Kind string // "layout" or "template"
	Name string
	Icon string // template icon (e.g. "⧉"), empty for layouts
	Meta string // "3 tabs  May 28" or "editor+git+term"
}

// PopResult holds the user's selection from the picker.
type PopResult struct {
	Item      PopItem
	Cancelled bool
}

// PopModel is a lightweight bubbletea picker for crex pop.
type PopModel struct {
	items    []PopItem
	filtered []PopItem
	filter   string
	cursor   int
	chosen   *PopItem
	quitting bool
	width    int
	height   int
}

// NewPopModel creates a picker with the given items.
func NewPopModel(items []PopItem, width, height int) *PopModel {
	m := &PopModel{
		items:  items,
		width:  width,
		height: height,
	}
	m.applyFilter()
	return m
}

func (m *PopModel) Init() tea.Cmd {
	return nil
}

func (m *PopModel) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.WindowSizeMsg:
		m.width = msg.Width
		m.height = msg.Height
		return m, nil

	case tea.KeyMsg:
		switch msg.Type {
		case tea.KeyEsc, tea.KeyCtrlC:
			m.quitting = true
			return m, tea.Quit

		case tea.KeyEnter:
			if len(m.filtered) > 0 && m.cursor < len(m.filtered) {
				chosen := m.filtered[m.cursor]
				m.chosen = &chosen
			}
			m.quitting = true
			return m, tea.Quit

		case tea.KeyUp:
			m.cursorUp()
			return m, nil

		case tea.KeyDown:
			m.cursorDown()
			return m, nil

		case tea.KeyBackspace:
			if len(m.filter) > 0 {
				m.filter = m.filter[:len(m.filter)-1]
				m.applyFilter()
				m.cursor = 0
			}
			return m, nil

		default:
			s := msg.String()
			// Number keys 1-9 for instant selection.
			if len(s) == 1 && s[0] >= '1' && s[0] <= '9' {
				n := int(s[0] - '0')
				if chosen := m.selectByNumber(n); chosen != nil {
					m.chosen = chosen
					m.quitting = true
					return m, tea.Quit
				}
			}
			// Printable characters → filter.
			if len(s) == 1 && s[0] >= ' ' && s[0] <= '~' {
				m.filter += s
				m.applyFilter()
				m.cursor = 0
			}
			return m, nil
		}
	}
	return m, nil
}

// Chosen returns the selected item, or nil if cancelled.
func (m *PopModel) Chosen() *PopItem {
	return m.chosen
}

func (m *PopModel) View() string {
	if m.quitting {
		return ""
	}

	headerStyle := lipgloss.NewStyle().Bold(true).Foreground(lipgloss.Color("6"))
	dimStyle := lipgloss.NewStyle().Foreground(lipgloss.Color("8"))
	selectedStyle := lipgloss.NewStyle().Bold(true).Foreground(lipgloss.Color("2"))
	numStyle := lipgloss.NewStyle().Foreground(lipgloss.Color("8"))

	var b strings.Builder

	// Filter line.
	prompt := "  crex > "
	if m.filter != "" {
		b.WriteString(prompt + m.filter + "\n")
	} else {
		b.WriteString(prompt + dimStyle.Render("type to filter...") + "\n")
	}
	b.WriteString("\n")

	// Group filtered items by kind.
	var layouts, templates []PopItem
	for _, item := range m.filtered {
		switch item.Kind {
		case "layout":
			layouts = append(layouts, item)
		case "template":
			templates = append(templates, item)
		}
	}

	globalIdx := 0

	// Layouts section.
	if len(layouts) > 0 {
		b.WriteString("  " + headerStyle.Render("LAYOUTS") + "\n")
		for _, item := range layouts {
			prefix := "   "
			if globalIdx == m.cursor {
				prefix = "  >"
			}
			num := numStyle.Render(fmt.Sprintf("[%d]", globalIdx+1))
			name := item.Name
			if globalIdx == m.cursor {
				name = selectedStyle.Render(name)
			}
			b.WriteString(fmt.Sprintf("%s %s  %s  %s\n", prefix, num, name, dimStyle.Render(item.Meta)))
			globalIdx++
		}
		b.WriteString("\n")
	}

	// Templates section.
	if len(templates) > 0 {
		b.WriteString("  " + headerStyle.Render("TEMPLATES") + "\n")
		for _, item := range templates {
			prefix := "   "
			if globalIdx == m.cursor {
				prefix = "  >"
			}
			num := numStyle.Render(fmt.Sprintf("[%d]", globalIdx+1))
			icon := item.Icon
			if icon == "" {
				icon = " "
			}
			name := item.Name
			if globalIdx == m.cursor {
				name = selectedStyle.Render(name)
			}
			b.WriteString(fmt.Sprintf("%s %s  %s  %s  %s\n", prefix, num, icon, name, dimStyle.Render(item.Meta)))
			globalIdx++
		}
		b.WriteString("\n")
	}

	if len(m.filtered) == 0 {
		b.WriteString("  " + dimStyle.Render("no matches") + "\n\n")
	}

	// Footer.
	b.WriteString("  " + dimStyle.Render("↵ launch · type to filter · esc quit") + "\n")

	return b.String()
}

// applyFilter filters items by fuzzy-matching the filter string against names.
func (m *PopModel) applyFilter() {
	if m.filter == "" {
		m.filtered = make([]PopItem, len(m.items))
		copy(m.filtered, m.items)
		return
	}
	lower := strings.ToLower(m.filter)
	m.filtered = nil
	for _, item := range m.items {
		if strings.Contains(strings.ToLower(item.Name), lower) ||
			strings.Contains(strings.ToLower(item.Meta), lower) {
			m.filtered = append(m.filtered, item)
		}
	}
}

func (m *PopModel) cursorUp() {
	if m.cursor > 0 {
		m.cursor--
	}
}

func (m *PopModel) cursorDown() {
	if m.cursor < len(m.filtered)-1 {
		m.cursor++
	}
}

func (m *PopModel) selectByNumber(n int) *PopItem {
	idx := n - 1 // 1-based to 0-based
	if idx >= 0 && idx < len(m.filtered) {
		item := m.filtered[idx]
		return &item
	}
	return nil
}

package cmd

import (
	"fmt"
	"os"
	"path/filepath"
	"sort"
	"time"

	tea "github.com/charmbracelet/bubbletea"
	"github.com/drolosoft/cmux-resurrect/internal/config"
	"github.com/drolosoft/cmux-resurrect/internal/gallery"
	"github.com/drolosoft/cmux-resurrect/internal/model"
	"github.com/drolosoft/cmux-resurrect/internal/orchestrate"
	"github.com/drolosoft/cmux-resurrect/internal/persist"
	"github.com/drolosoft/cmux-resurrect/internal/tui"
	"github.com/spf13/cobra"
)

var popLast bool

var popCmd = &cobra.Command{
	Use:   "pop [name] [path]",
	Short: "Quick workspace launcher — Spotlight for your terminal",
	Long: `Open a picker to launch saved layouts or templates instantly.

With no arguments, shows a filterable picker. With arguments, launches directly:
  crex pop morning        restore a saved layout
  crex pop ide .          apply a template to a directory
  crex pop --last         restore the most recently saved layout`,
	Args:              cobra.MaximumNArgs(2),
	RunE:              runPop,
	ValidArgsFunction: completePopArgs,
}

func init() {
	popCmd.Flags().BoolVar(&popLast, "last", false, "restore the most recently saved layout")
	rootCmd.AddCommand(popCmd)
}

func runPop(cmd *cobra.Command, args []string) error {
	store, err := newStore()
	if err != nil {
		return fmt.Errorf("open store: %w", err)
	}

	// --last: restore most recent layout.
	if popLast {
		return popRestoreLast(store)
	}

	// With args: direct launch.
	if len(args) > 0 {
		return popDirect(store, args)
	}

	// No args: show picker.
	return popPicker(store)
}

func popRestoreLast(store persist.Store) error {
	metas, err := store.List()
	if err != nil {
		return err
	}
	if len(metas) == 0 {
		return fmt.Errorf("no saved layouts")
	}
	// Find most recent by SavedAt.
	sort.Slice(metas, func(i, j int) bool {
		return metas[i].SavedAt.After(metas[j].SavedAt)
	})
	return doRestore(store, metas[0].Name)
}

func popDirect(store persist.Store, args []string) error {
	name := args[0]

	// Try as saved layout first.
	if store.Exists(name) {
		return doRestore(store, name)
	}

	// Try as template.
	if tmpl, ok := gallery.Get(name); ok {
		cwd := "."
		if len(args) >= 2 {
			cwd = args[1]
		}
		return doTemplateUse(tmpl, cwd)
	}

	return fmt.Errorf("%q is not a saved layout or template name", name)
}

func popPicker(store persist.Store) error {
	items, err := buildPopItems(store)
	if err != nil {
		return err
	}
	if len(items) == 0 {
		return fmt.Errorf("no layouts or templates found — run 'crex save' or 'crex template list'")
	}

	m := tui.NewPopModel(items, 80, 24)
	p := tea.NewProgram(m, tea.WithAltScreen())
	finalModel, err := p.Run()
	if err != nil {
		return err
	}

	pm := finalModel.(*tui.PopModel)
	chosen := pm.Chosen()
	if chosen == nil {
		return nil // cancelled
	}

	switch chosen.Kind {
	case "layout":
		return doRestore(store, chosen.Name)
	case "template":
		return doTemplateUse(nil, ".") // resolve below
	}
	return nil
}

func buildPopItems(store persist.Store) ([]tui.PopItem, error) {
	var items []tui.PopItem

	// Saved layouts.
	metas, err := store.List()
	if err != nil {
		return nil, err
	}
	// Sort by most recent first.
	sort.Slice(metas, func(i, j int) bool {
		return metas[i].SavedAt.After(metas[j].SavedAt)
	})
	for _, meta := range metas {
		age := formatAge(meta.SavedAt)
		items = append(items, tui.PopItem{
			Kind: "layout",

func formatAge(t time.Time) string {
	d := time.Since(t)
	switch {
	case d < time.Hour:
		return "just now"
	case d < 24*time.Hour:
		return fmt.Sprintf("%dh ago", int(d.Hours()))
	case d < 7*24*time.Hour:
		return fmt.Sprintf("%dd ago", int(d.Hours()/24))
	default:
		return t.Format("Jan 2")
	}
}
